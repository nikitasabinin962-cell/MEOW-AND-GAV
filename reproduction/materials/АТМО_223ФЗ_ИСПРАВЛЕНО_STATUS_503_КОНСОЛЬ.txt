 function arrayCandidates(root){
  const found=[];

   if(Array.isArray(x)){

       (x.length===view.length?4:0)+ (x.some(z=>/ИЗ223-20\d{2}-\d+/i.test(JSON.stringify(z)))?25:0);
     found.push({path,arr:x,score});

    return;

   if(!x||typeof x!=='object'||d>5)return;
   for(const [k,v] of Object.entries(x))visit(v,[...path,k],d+1);
  }
  visit(root);return found.sort((a,b)=>b.score-a.score);

 function learnPath(items,expectedField){

  const good=pairs.filter(x=>x.expect!==undefined);

     filter,source:'АТМО 223-ФЗ / Договор заключён',pageCount:state.totalPages,saved:state.saved,
     overlaps:state.overlaps,files:state.part-1,signaturesVerified:false,

    },null,2));
    stat('ГОТОВО. Все страницы 223-ФЗ пройдены, файлы сохранены отдельно.');
   }
  }catch(e){

   try{await savePart(true);}catch(saveError){console.error('Не удалось записать буфер, он сохранён в IndexedDB:',saveError);}
   stat('ОСТАНОВКА: '+e.message);
